    <script>
        // Set Current Date
        const dateOptions = { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' };
        document.getElementById('liveDate').textContent = new Date().toLocaleDateString('id-ID', dateOptions);

        // Toggle Open/Close Letter
        function toggleEnvelope() {
            const closedState = document.getElementById('cardClosedState');
            const openState = document.getElementById('cardOpenState');
            
            if (openState.classList.contains('hidden')) {
                closedState.classList.add('hidden');
                openState.classList.remove('hidden');
            } else {
                openState.classList.add('hidden');
                closedState.classList.remove('hidden');
            }
        }

        // Micro-Notes / Affirmations
        const affirmations = [
            "Tarik napas dalam-dalam. Satu demi satu soal pasti bisa kamu selesaikan.",
            "Kamu sudah belajar, selebihnya biar ketenangan dan keberuntungan yang bekerja.",
            "Semoga materi ujian yang keluar pas banget sama yang kamu ingat!",
            "Jangan buru-buru, kerjakan dengan tenang. Doaku menyertaimu dari sini.",
            "Apapun hasilnya nanti, kamu tetap cantiknya aku yang selalu bikin bangga."
        ];

        function generateMicroAffirmation() {
            const box = document.getElementById('affirmationBox');
            const randomIndex = Math.floor(Math.random() * affirmations.length);
            box.textContent = `"${affirmations[randomIndex]}"`;
            box.classList.remove('hidden');
        }

        // Direct WhatsApp Reply Link
        function sendWhatsAppReply() {
            const defaultText = "Pagi juga! Makasih ya ucapan semangatnya, ini aku mau siap-siap sekolah/ujian dulu. ❤️";
            const encodedText = encodeURIComponent(defaultText);
            window.open(`https://wa.me/?text=${encodedText}`, '_blank');
        }
